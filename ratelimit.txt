package com.bel.security.controller;

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RestController;

import com.bel.security.utils.BucketCacheService;

import io.github.bucket4j.Bucket;

@RestController
public class RateLimitController {

    @Autowired
    private BucketCacheService bucketCacheService;

    @GetMapping("/admin/buckets")
    public List<BucketData> getAllBuckets() {

        Map<String, Bucket> buckets = bucketCacheService.getAllBuckets();

        return buckets.entrySet()
                .stream()
                .map(entry -> new BucketData(
                        entry.getKey(),
                        getEndpoint(entry.getKey()),
                        entry.getValue().getAvailableTokens()
                ))
                .collect(Collectors.toList());
    }

    private String getEndpoint(String key) {
        if (key.contains(":")) {
            return key.split(":")[1];
        }
        return "unknown";
    }

    public static class BucketData {

        private String key;
        private String endpoint;
        private long availableTokens;

        public BucketData(String key, String endpoint, long availableTokens) {
            this.key = key;
            this.endpoint = endpoint;
            this.availableTokens = availableTokens;
        }

        public String getKey() {
            return key;
        }

        public String getEndpoint() {
            return endpoint;
        }

        public long getAvailableTokens() {
            return availableTokens;
        }
    }
}

package com.bel.security.utils;

import java.time.Duration;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;

import io.github.bucket4j.Bandwidth;
import io.github.bucket4j.Bucket;

@Component
//Service that manages all rate-limiting buckets
public class BucketCacheService {


	    /**
	     * Map defining rate limits per endpoint.
	     * Key: endpoint name (e.g., "auth", "public_auth")
	     * Value: Bandwidth object from Bucket4j defining:
	     *   - capacity: maximum number of requests allowed
	     *   - refillIntervally: how many tokens are added back per interval
	     */
	    private final Map<String, Bandwidth> endpointLimits = Map.of(
	        "public_auth",
	            Bandwidth.builder()
	                .capacity(5)                        // maximum 5 requests
	                .refillIntervally(5, Duration.ofMinutes(1)) // refill 5 tokens every 1 minute
	                .build(),
	        "auth",
	            Bandwidth.builder()
	                .capacity(10)                       // maximum 10 requests
	                .refillIntervally(10, Duration.ofMinutes(1)) // refill 10 tokens every 1 minute
	                .build()
	    );

	    // Optional: you could have used a simple ConcurrentHashMap to store buckets, but
	    // we are using a more advanced cache (Caffeine) below.
	    // private final ConcurrentHashMap<String, Bucket> bucketMap = new ConcurrentHashMap<>();

	    /**
	     * Caffeine cache storing all buckets.
	     * Key: a combination of IP + endpoint (e.g., "192.168.0.1:auth")
	     * Value: Bucket4j Bucket object that tracks token consumption
	     * Features:
	     *   - maximumSize(10_000): cache will not store more than 10k buckets
	     *   - expireAfterAccess(Duration.ofMinutes(10)): removes buckets that haven't
	     *     been used in the last 10 minutes
	     */
	    private final Cache<String, Bucket> bucketCache = Caffeine.newBuilder()
	        .maximumSize(10_000)
	        .expireAfterAccess(Duration.ofMinutes(10))
	        .build();

	    /**
	     * Get or create a bucket for a specific key and endpoint.
	     * 
	     * @param key      unique key (e.g., IP address)
	     * @param endpoint endpoint name (e.g., "auth", "public_auth")
	     * @return Bucket4j Bucket object
	     *
	     * Logic:
	     *   - If a bucket already exists in the cache, return it.
	     *   - If not, create a new bucket with the rate limits of the endpoint and
	     *     store it in the cache.
	     */
	    public Bucket getBucket(String key, String endpoint) {
	        return bucketCache.get(key, k -> createBucket(endpoint));
	    }

	    /**
	     * Returns all buckets currently stored in the cache.
	     * Useful for monitoring or exposing via an API.
	     *
	     * @return map of all buckets
	     */
	    public Map<String, Bucket> getAllBuckets() {
	        return bucketCache.asMap();
	    }

	    /**
	     * Create a new bucket for a given endpoint using its rate limit.
	     * 
	     * @param endpoint endpoint name
	     * @return new Bucket4j Bucket
	     *
	     * Logic:
	     *   - Look up the endpointLimits map to get the rate limit.
	     *   - If the endpoint is not configured, use a default limit of 20 requests
	     *     per minute.
	     *   - Build and return the Bucket object.
	     */
	    private Bucket createBucket(String endpoint) {
	        Bandwidth limit = endpointLimits.getOrDefault(
	            endpoint,
	            Bandwidth.builder()
	                .capacity(20)                    // default 20 requests
	                .refillIntervally(20, Duration.ofMinutes(1)) // default refill
	                .build()
	        );
	        return Bucket.builder().addLimit(limit).build();
	    }

	    /**
	     * Scheduled task to clear all buckets from the cache every 20 minutes.
	     * 
	     * This is optional. It ensures that old buckets that might not be used
	     * anymore are cleared, freeing memory.
	     */
	    @Scheduled(fixedRate = 20 * 60 * 1000) // runs every 20 minutes
	    public void clearAllBuckets() {
	        bucketCache.invalidateAll(); // remove all buckets
	        System.out.println("All buckets cleared at " + java.time.LocalDateTime.now());
	    }
}

	private String getBucketKey(String path) {
		if (path.startsWith("/public/auth"))
			return "public_auth";

		return "default";

	private boolean isPublicApi(String path) {
		System.out.println("path in ispublic =" + path);
		return path.startsWith("/public") || path.contains("/auth") || path.contains("/swagger-ui.html")
				|| path.contains("/swagger-ui") || path.contains("/v3/api-docs") || path.contains("/actuator")
				|| path.contains("/favicon.ico");
	}

	/** Log all request headers */
	private void logRequestHeaders(HttpServletRequest request) {
		Enumeration<String> headerNames = request.getHeaderNames();
		if (headerNames != null) {
			while (headerNames.hasMoreElements()) {
				String name = headerNames.nextElement();
				logger.info("Header {} = {}", name, request.getHeader(name));
			}
		}
	}

	/** Get the real client IP, considering X-Forwarded-For */
	private String getClientIP(HttpServletRequest request) {
		String xfHeader = request.getHeader("X-Forwarded-For");
		if (xfHeader == null)
			return request.getRemoteAddr();
		return xfHeader.split(",")[0];
	}

    server {
        listen 8081;  # Nginx port
        server_name localhost;

        # ---------------------------
        # /api/public/auth/** → rate limit 5r/m
        # ---------------------------
        location /api/public/auth {
            limit_req zone=public_auth_zone burst=2 nodelay;

            proxy_pass http://127.0.0.1:8010;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # ---------------------------
        # /api/auth/** → rate limit 10r/m
        # ---------------------------
        location /api/auth {
            limit_req zone=auth_zone burst=5 nodelay;

            proxy_pass http://127.0.0.1:8010;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # ---------------------------
        # Catch-all for other endpoints (no rate limit)
        # ---------------------------
        location / {
            proxy_pass http://127.0.0.1:8010;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # ---------------------------
        # Custom 429 response (rate limit exceeded)
        # ---------------------------
        error_page 429 = @rate_limit_exceeded;

        location @rate_limit_exceeded {
            default_type text/plain;
            return 429 "Too many requests. Please wait a moment and try again.\n";
        }

        # ---------------------------
        # Optional security headers
        # ---------------------------
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;
